# =============================================================================
#  Diamond Engine — native core (libengine.so)
# =============================================================================
#  Language split
#    * C99   — engine ABI, math helpers, scene storage, deflate, logging,
#              block-logic execution, LuaC host glue (all `extern "C"`).
#    * C++23 — OBJ parser, .diamond serializer, render backends, JNI bridge.
#
#  Weight policy
#    * -Os everywhere, -fno-exceptions -fno-rtti for C++ (no unwind tables,
#      no typeinfo in the .so), -fvisibility=hidden + explicit C exports.
#    * No libc++ streams, no <iostream>, no <regex>, no <filesystem>.
#
#  Options (all overridable with -D… from app/build.gradle):
#    DIAMOND_USE_LUAC      1 = vendored LuaC under third_party/luac is compiled
#    DIAMOND_HAVE_LUAC     defined by this file when LuaC sources exist
#    DIAMOND_SPV_PRESET    SPIR-V shaders were precompiled in CI
#    DIAMOND_ENABLE_VULKAN 1 = build the Vulkan backend
#    DIAMOND_STRICT        treat warnings as errors (CI style check)
# =============================================================================
cmake_minimum_required(VERSION 3.22.1)

project(DiamondEngine
        VERSION 0.1.0
        DESCRIPTION "Ultra-light 2D/2.5D/3D engine + scene editor for Android"
        LANGUAGES C CXX)

# -----------------------------------------------------------------------------
#  Options
# -----------------------------------------------------------------------------
option(DIAMOND_USE_LUAC     "Compile the vendored LuaC (nightly branch)" ON)
option(DIAMOND_ENABLE_VULKAN "Build the experimental Vulkan backend" ON)
# The Vulkan ABI headers only exist from NDK r24 on; the backend is skipped
# automatically on older toolchains instead of failing the whole build.
option(DIAMOND_STRICT       "Warnings are errors" OFF)

set(DIAMOND_SPV_PRESET "${DIAMOND_SPV_PRESET}" CACHE STRING "SPIR-V shaders precompiled in CI (0/1)")
set(DIAMOND_SHADER_INCLUDE "${DIAMOND_SHADER_INCLUDE}" CACHE STRING
    "Directory holding shader_bin_data.h generated by tools/compile_shaders.py")

if(NOT DEFINED DIAMOND_LUAC_REF)
    set(DIAMOND_LUAC_REF "arena/b210b158-luac")
endif()

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(DIAMOND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(DIAMOND_LUAC_DIR "${DIAMOND_ROOT}/third_party/luac")

message(STATUS "Diamond: NDK ${ANDROID_NDK_RELEASE} / ABI ${ANDROID_ABI} / API ${ANDROID_PLATFORM_LEVEL}")
message(STATUS "Diamond: LuaC ref '${DIAMOND_LUAC_REF}' requested, use_luac=${DIAMOND_USE_LUAC}")

# -----------------------------------------------------------------------------
#  LuaC discovery — the sources are cloned by CI into third_party/luac.
#  When they are missing (or their layout changes in a future nightly) the
#  engine still builds: blocks logic then runs on the built-in native executor
#  and the UI states clearly that scripting is disabled.
# -----------------------------------------------------------------------------
set(DIAMOND_HAVE_LUAC 0)
if(DIAMOND_USE_LUAC)
    if(EXISTS "${DIAMOND_LUAC_DIR}/src/luac.c"
       AND EXISTS "${DIAMOND_LUAC_DIR}/src/luac.h"
       AND EXISTS "${DIAMOND_LUAC_DIR}/src/runtime.h")
        set(DIAMOND_HAVE_LUAC 1)
    else()
        message(WARNING
                "Diamond: LuaC sources not found in ${DIAMOND_LUAC_DIR}.\n"
                "         Building with USE_BUILTIN_FALLBACK=1 (native block executor only).\n"
                "         CI must run the 'Fetch LuaC nightly revision' step first.")
    endif()
else()
    message(STATUS "Diamond: DIAMOND_USE_LUAC=OFF -> USE_BUILTIN_FALLBACK=1")
endif()

# -----------------------------------------------------------------------------
#  Source lists
# -----------------------------------------------------------------------------
set(DIAMOND_C_SOURCES
        core/dlog.c
        core/dtime.c
        core/darena.c
        core/dstr.c
        core/deflate.c
        core/dhash.c
        logic/logic_exec.c
        logic/logic_debug.c
        logic/lua_host.c)

set(DIAMOND_CXX_SOURCES
        scene/math.cpp
        scene/scene.cpp
        scene/obj_loader.cpp
        format/diamond_format.cpp
        logic/logic_blocks.cpp
        logic/logic_codegen.cpp
        render/renderer.cpp
        render/camera.cpp
        render/gl_context.cpp
        render/gl_program.cpp
        render/immediate.cpp
        render/gl_backend.cpp
        render/gl2_backend.cpp
        render/hybrid_backend.cpp
        render/software_raster.cpp
        platform/native_window.cpp
        platform/file_io.cpp
        jni/jni_bridge.cpp
        api/diamond_api.cpp)

# The Vulkan sources are always compiled when the option is on. Without
# <vulkan/vulkan.h> they provide the "unavailable" backend, which fails attach()
# with a real message, so a missing header degrades the app instead of the build.
set(DIAMOND_VULKAN_ENABLED 0)
if(DIAMOND_ENABLE_VULKAN)
    set(DIAMOND_VULKAN_ENABLED 1)
    list(APPEND DIAMOND_CXX_SOURCES render/vulkan_backend.cpp platform/vulkan_surface.cpp)
    include(CheckIncludeFileCXX)
    check_include_file_cxx("vulkan/vulkan.h" DIAMOND_HAVE_VK_HEADER)

    # The loader comes in two shapes. With the NDK's libvulkan.so, -lvulkan
    # resolves. With a staged platform tree whose lib dir is outside the default
    # search path the import library is missing even though the header is there,
    # and linking would fail. In that case the backend still works at runtime:
    # every dvulkan_* path dlopen()s libvulkan.so and resolves entry points
    # through vkGetInstanceProcAddr, so no link against the loader is needed.
    find_library(DIAMOND_VULKAN_LIBRARY vulkan)
    if(DIAMOND_HAVE_VK_HEADER AND DIAMOND_VULKAN_LIBRARY)
        message(STATUS "Diamond: Vulkan backend enabled (headers + ${DIAMOND_VULKAN_LIBRARY})")
        set(DIAMOND_LINK_VULKAN 1)
    elseif(DIAMOND_HAVE_VK_HEADER)
        message(STATUS
                "Diamond: Vulkan backend enabled, headers only — libvulkan.so is not "
                "in the link path, the loader is dlopen()ed at runtime instead")
        set(DIAMOND_LINK_VULKAN 0)
    else()
        message(STATUS "Diamond: Vulkan backend compiled in stub mode (no vulkan/vulkan.h)")
        set(DIAMOND_LINK_VULKAN 0)
    endif()
endif()

# -----------------------------------------------------------------------------
#  Static library: flat C core (C99)
# -----------------------------------------------------------------------------
add_library(diamond_c STATIC ${DIAMOND_C_SOURCES})
set_target_properties(diamond_c PROPERTIES
        C_STANDARD 99
        C_STANDARD_REQUIRED ON
        C_VISIBILITY_PRESET hidden
        POSITION_INDEPENDENT_CODE ON
        LINKER_LANGUAGE C)

target_include_directories(diamond_c
        PUBLIC include
        PRIVATE . include/diamond)

# logic/lua_host.c is compiled here but #includes "luac.h" when LuaC is
# present, so diamond_c needs LuaC's headers too. Linking diamond_luac to it
# would drag the whole LuaC library into every consumer of the C core, so only
# the include path is shared.
if(DIAMOND_HAVE_LUAC)
    target_include_directories(diamond_c PRIVATE "${DIAMOND_LUAC_DIR}/src")
endif()

target_compile_definitions(diamond_c PRIVATE
        DIAMOND_HAVE_LUAC=${DIAMOND_HAVE_LUAC}
        DIAMOND_ENABLE_VULKAN=$<BOOL:${DIAMOND_ENABLE_VULKAN}>
        DIAMOND_SPV_PRESET=${DIAMOND_SPV_PRESET}
        DIAMOND_LUAC_REF="${DIAMOND_LUAC_REF}"
        DIAMOND_ENGINE_VERSION="0.1.0")

target_compile_options(diamond_c PRIVATE
        -Os
        -fno-exceptions
        -fno-rtti
        -fno-asynchronous-unwind-tables
        -fomit-frame-pointer
        -fvisibility=hidden
        -ffunction-sections
        -fdata-sections
        -Wall
        -Wextra
        -Wno-unused-parameter
        -Wno-sign-compare
        $<$<BOOL:${DIAMOND_STRICT}>:-Werror>)

# -----------------------------------------------------------------------------
#  Static library: C++23 half (scene, format, render, JNI)
# -----------------------------------------------------------------------------
add_library(diamond_cxx STATIC ${DIAMOND_CXX_SOURCES})
set_target_properties(diamond_cxx PROPERTIES
        CXX_STANDARD 23
        CXX_STANDARD_REQUIRED ON
        CXX_EXTENSIONS OFF
        CXX_VISIBILITY_PRESET hidden
        VISIBILITY_INLINES_HIDDEN ON
        POSITION_INDEPENDENT_CODE ON)

target_include_directories(diamond_cxx
        PUBLIC include
        PRIVATE . include/diamond render)

# Shaders are baked by tools/compile_shaders.py before CMake runs (see the CI
# workflow). Without the generated header the Vulkan backend still links and
# reports "SPIR-V shaders were not baked in" instead of drawing.
if(DIAMOND_SPV_PRESET AND EXISTS "${DIAMOND_SHADER_INCLUDE}/shader_bin_data.h")
    target_include_directories(diamond_cxx PRIVATE "${DIAMOND_SHADER_INCLUDE}")
    target_compile_definitions(diamond_cxx PRIVATE DIAMOND_HAVE_SPV=1)
endif()

target_compile_definitions(diamond_cxx PRIVATE
        DIAMOND_HAVE_LUAC=${DIAMOND_HAVE_LUAC}
        DIAMOND_ENABLE_VULKAN=$<BOOL:${DIAMOND_ENABLE_VULKAN}>
        DIAMOND_SPV_PRESET=${DIAMOND_SPV_PRESET}
        DIAMOND_LUAC_REF="${DIAMOND_LUAC_REF}"
        DIAMOND_ENGINE_VERSION="0.1.0")

target_compile_options(diamond_cxx PRIVATE
        -Os
        -fno-exceptions
        -fno-rtti
        -fno-threadsafe-statics
        -fno-asynchronous-unwind-tables
        -fomit-frame-pointer
        -fvisibility=hidden
        -fvisibility-inlines-hidden
        -ffunction-sections
        -fdata-sections
        -Wall
        -Wextra
        -Wno-unused-parameter
        -Wno-sign-compare
        -Wno-missing-field-initializers
        $<$<BOOL:${DIAMOND_STRICT}>:-Werror>)

# -----------------------------------------------------------------------------
#  LuaC (vendored nightly branch) — compiled as C99 into a static library.
# -----------------------------------------------------------------------------
if(DIAMOND_HAVE_LUAC)
    add_library(diamond_luac STATIC "${DIAMOND_LUAC_DIR}/src/luac.c")
    set_target_properties(diamond_luac PROPERTIES
            C_STANDARD 99
            C_STANDARD_REQUIRED ON
            C_EXTENSIONS ON
            POSITION_INDEPENDENT_CODE ON
            LINKER_LANGUAGE C)

    target_include_directories(diamond_luac PUBLIC "${DIAMOND_LUAC_DIR}/src")
    target_compile_definitions(diamond_luac PRIVATE
            LUAC_LIBRARY_ONLY=1
            LUAC_EMBEDDED_RUNTIME=1
            LUAC_RUNTIME_PATH="runtime.h")
    target_compile_options(diamond_luac PRIVATE
            -Os
            -fno-exceptions
            -fno-rtti
            -ffunction-sections
            -fdata-sections
            -Wno-unused-parameter
            -Wno-sign-compare
            # LuaC is vendored verbatim; its warnings must not break our build.
            -Wno-everything)
    message(STATUS "Diamond: LuaC compiled from ${DIAMOND_LUAC_DIR}/src/luac.c")
endif()

# -----------------------------------------------------------------------------
#  The one and only shipped shared object
# -----------------------------------------------------------------------------
# The shared object has no translation unit of its own: it links the three
# static halves plus the export map below. CMake needs at least one source, so
# jni_anchor.cpp holds JNI_OnLoad/JNI_OnUnload only.
#
# All three static libraries are linked with --whole-archive. Without it the
# linker only pulls in objects that the anchor defines a *reference* to, and
# nothing references the JNI entry points: they are called by the Java VM
# through dlsym(), so Java_com_diamond_engine_jni_NativeBridge_init() never
# reaches the dynamic symbol table and the first System.loadLibrary() call dies
# with UnsatisfiedLinkError. Verified by inspecting nm -D on a linked .so.
add_library(engine SHARED
        ${DIAMOND_ROOT}/jni/jni_anchor.cpp)

# Every static half this .so ships, in link order. The whole-archive wrapper
# keeps all of their objects, so the JNI entry points (which nothing references
# at link time) land in the dynamic symbol table.
set(DIAMOND_STATIC_HALVES diamond_cxx diamond_c)
if(DIAMOND_HAVE_LUAC)
    list(APPEND DIAMOND_STATIC_HALVES diamond_luac)
endif()

set_target_properties(engine PROPERTIES
        OUTPUT_NAME engine
        C_STANDARD 99
        POSITION_INDEPENDENT_CODE ON)

target_include_directories(engine PUBLIC include)

target_compile_options(engine PRIVATE
        -Os
        -fvisibility=hidden
        -ffunction-sections
        -fdata-sections)

# Order matters and plain flags are kept verbatim by CMake, so the whole-archive
# bracket can live in the same call: the static halves come first (they are what
# needs android/log/EGL), the platform import libraries after them. Wrapping
# android/log/EGL themselves would be wrong — they are import stubs, not archives.
target_link_libraries(engine PRIVATE
        "-Wl,--whole-archive"
        ${DIAMOND_STATIC_HALVES}
        "-Wl,--no-whole-archive"
        android
        log
        m
        dl
        EGL
        GLESv3
        # AndroidBitmap_* (jni_bridge.cpp uploads a Bitmap as a sprite texture)
        # lives in libjnigraphics, not in libandroid.
        jnigraphics)

if(DIAMOND_LINK_VULKAN)
    # libvulkan.so only exports the entry points; the loader does the rest.
    # Not linked when the import library is absent — see the probe above.
    target_link_libraries(engine PRIVATE "${DIAMOND_VULKAN_LIBRARY}")
endif()

# 16 KiB page size — mandatory for Android 15+ devices with 16 KiB kernels,
# harmless on 4 KiB systems.
#
# --gc-sections keeps the size down even with --whole-archive: an object is
# pulled in, but sections that nothing reaches are still discarded.
target_link_options(engine PRIVATE
        "-Wl,--version-script=${DIAMOND_ROOT}/libengine.map"
        "-Wl,-z,max-page-size=16384"
        "-Wl,--gc-sections")

# -----------------------------------------------------------------------------
#  Build report (printed into the CI log for every ABI)
# -----------------------------------------------------------------------------
add_custom_command(TARGET engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E echo
        "Diamond: libengine.so built for ${ANDROID_ABI} (LuaC=${DIAMOND_HAVE_LUAC}, Vulkan=${DIAMOND_ENABLE_VULKAN}, SPV=${DIAMOND_SPV_PRESET})"
        VERBATIM)
